func test(_ result: Data?, label: String) {
    let store = LegacyTaskStore()
    let initial = ReleaseRequest.starts
    let now = Date(timeIntervalSince1970: 1_700_000_000)
    store.checkForUpdates(now: now)
    check(ReleaseRequest.starts == initial + 1, "macOS 0.3.7 \(label): checks immediately")
    store.releaseRequest?.finish(result)
    RunLoop.main.run(until: Date().addingTimeInterval(0.05))
    check(store.releaseRequest == nil, "macOS 0.3.7 \(label): original callback finishes")
    store.checkForUpdates(now: now.addingTimeInterval(86399))
    check(ReleaseRequest.starts == initial + 1, "macOS 0.3.7 \(label): not due before 24 hours")
    store.checkForUpdates(now: now.addingTimeInterval(86400))
    check(ReleaseRequest.starts == initial + 2, "macOS 0.3.7 \(label): retries after 24 hours")
}
